function closeDedicatedGame(){ const screen = document.getElementById("dedicatedGameScreen"); if(!screen) return; if(gameTimer){ clearInterval(gameTimer); gameTimer = null; } if(gameAnimation){ cancelAnimationFrame(gameAnimation); gameAnimation = null; } screen.classList.remove("active"); /* Return to Games screen */ if(location.hash !== "#games"){ history.pushState( { aifaPage:"games" }, "", "#games" ); } showPage("games", null, false); }
